# SSH Config untuk QR-Tunai Server
# File ini akan disimpan di ~/.ssh/config

# QR-Tunai Production Server
Host qr-tunai-server
    HostName YOUR_SERVER_IP
    User YOUR_SERVER_USERNAME
    Port 22
    IdentityFile ~/.ssh/id_rsa
    ForwardAgent yes
    ServerAliveInterval 60
    ServerAliveCountMax 3
